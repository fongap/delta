# VS_VERSION_INFO version resource for the bundled sidecar (delta-server.exe).

    filevers=(0, 2, 0, 0),
    prodvers=(0, 2, 0, 0),
    mask=0x3F,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          '040904B0',
          [
            StringStruct('CompanyName', 'Delta'),
            StringStruct('FileDescription', 'Delta Server'),
            StringStruct('FileVersion', '0.2.0'),
            StringStruct('InternalName', 'delta-server'),
            StringStruct('OriginalFilename', 'delta-server.exe'),
            StringStruct('ProductName', 'Delta'),
            StringStruct('ProductVersion', '0.2.0'),
          ]
        )
      ]
    ),
    VarFileInfo([VarStruct('Translation', [0x409, 1200])])
  ]
)
